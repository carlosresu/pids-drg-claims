Update domain to redirect pids-drg-vm.resu.works to VM's external IP address
  Log in to your Cloudflare account.
  Select your domain (resu.works) from the dashboard.
  Navigate to the DNS settings:
  Click on the DNS tab.
  Add a new DNS record:
  Type: A
  Name: pids-drg-vm
  IPv4 address: Enter the External IP address of your GCP GCE VM.
  TTL: Automatic
  Proxy status: Enable or disable the proxy (orange cloud for enabled, grey cloud for disabled).
  Click Save to add the DNS record.
  
Verify DNS propagation: 
https://www.whatsmydns.net/#A/pids-drg-vm.resu.works

Once finished, do this:

=======CODE START=======
sudo nano /etc/nginx/sites-available/rstudio
========CODE END========

Paste the below into nginx config file:
server {
    listen 80;
    server_name pids-drg-vm.resu.works;

    location /.well-known/acme-challenge/ {
        root /var/www/html;
        allow all;
    }

    location / {
        proxy_pass http://localhost:8787;
        proxy_redirect http://localhost:8787/ $scheme://$host/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/pids-drg-vm.resu.works/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/pids-drg-vm.resu.works/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
}

=======CODE START=======
sudo ln -s /etc/nginx/sites-available/rstudio /etc/nginx/sites-enabled/
sudo rm /etc/nginx/sites-enabled/default

sudo systemctl restart nginx
========CODE END========
